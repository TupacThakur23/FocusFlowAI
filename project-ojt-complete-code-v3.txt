  try {
    console.log('🔍 Background: GET_SIDEBAR_STATE request received');

    return {
      success: true,

        isVisible: Boolean(sidebarState?.isVisible),
        isOpen: Boolean(sidebarState?.isOpen),
        isCollapsed: Boolean(sidebarState?.isCollapsed),
        activeView: sidebarState?.activeView || 'launcher',
        width: sidebarState?.width || 400
      }
    };
  } catch (error) {
    console.error('🚨 Background: Failed to get sidebar state:', error);
    return {
      success: false,
      error: error.message,
      state: {
        isVisible: false,
        isOpen: false,
        isCollapsed: false,
        activeView: 'launcher',
        width: 400
      }
    };
  }
});

// Lines 115-130: Added OPEN_SIDEBAR message handler
messageBus.onMessage('OPEN_SIDEBAR', async (message, sender) => {
  try {
    const activeTab = await this.getActiveTab();
    if (activeTab) {
      const response = await messageBus.sendMessage(activeTab.id, {
        type: 'OPEN_SIDEBAR',
        source: 'background'
      });
      return response;
    }
    return { success: false, error: 'No active tab' };
  } catch (error) {
    errorHandler.log(error, 'BackgroundService.OPEN_SIDEBAR');
    return { success: false, error: error.message };
  }
});

if (typeof chrome !== 'undefined' && chrome.runtime && chrome.runtime.onMessage) {
  chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
    console.log('🔍 Content script: Message received:', message);
    console.log('🔍 Content script: Message type:', message.type);
    
    if (message.type === 'OPEN_SIDEBAR') {
      console.log('🔍 Content script: OPEN_SIDEBAR message received - opening sidebar');
      setCollapsed(false);
      sendResponse({ success: true });
    } else if (message.type === 'CLOSE_SIDEBAR') {
      console.log('🔍 Content script: CLOSE_SIDEBAR message received - closing sidebar');
      setCollapsed(true);
      sendResponse({ success: true });
    } else {
      console.log('🔍 Content script: Unknown message type:', message.type);
    }
    return true; // Keep message channel open
  });
}

## PROBLEM SOLVING APPROACH

### Issues Addressed:
1. **Unclickable Buttons**: Fixed by removing disabled attributes and adding proper CSS z-index
2. **State Contract Mismatch**: Fixed by implementing state normalization in useSidebarState.js
3. **Missing Message Handlers**: Added proper message handlers in background.js and content.js
4. **Permissions Issues**: Added "tabs" permission to manifest.json
5. **CSS/Rendering Issues**: Fixed with inline styles and proper z-index values
6. **URL Construction**: Fixed proper Chrome extension URL construction
7. **Alert Interference**: Removed all alert() calls from button handlers

### Final Implementation Strategy:
- **Aide Button**: Opens Dashboard component in new tab using chrome.tabs.create()
- **Research Hub Button**: Opens Research Hub component in new tab using chrome.tabs.create()
- **No Popup Interference**: Both buttons work directly without popup switching
- **Simple Approach**: Direct Chrome API calls instead of complex message passing
- **Clean CSS**: Inline styles with proper z-index and pointer-events
